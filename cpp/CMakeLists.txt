cmake_minimum_required(VERSION 3.24)
project(OpenAxis VERSION 1.0.0 LANGUAGES CXX)
include(FetchContent)
option(OPENAXIS_BUILD_TESTS "Build SDK tests" ${PROJECT_IS_TOP_LEVEL})
option(OPENAXIS_BUILD_DEMO "Build the native reference viewer" OFF)
option(OPENAXIS_FETCH_DEPENDENCIES "Fetch missing pinned dependencies" ON)
find_package(nlohmann_json 3.11 QUIET)
if(NOT TARGET nlohmann_json::nlohmann_json)
  if(NOT OPENAXIS_FETCH_DEPENDENCIES)
    message(FATAL_ERROR "Install nlohmann_json or enable OPENAXIS_FETCH_DEPENDENCIES")
  endif()
  FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
    URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
  FetchContent_MakeAvailable(json)
endif()
add_subdirectory(third_party/ixwebsocket)
add_library(openaxis STATIC src/client.cpp src/connection_manager.cpp src/navigation.cpp src/diagnostics.cpp src/verify.cpp)
if(WIN32)
  target_link_libraries(openaxis PRIVATE bcrypt)
else()
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  target_link_libraries(openaxis PRIVATE OpenSSL::Crypto)
endif()
add_library(OpenAxis::openaxis ALIAS openaxis)
target_compile_features(openaxis PUBLIC cxx_std_17)
target_include_directories(openaxis PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_link_libraries(openaxis PUBLIC nlohmann_json::nlohmann_json)
target_link_libraries(openaxis PRIVATE openaxis_ixwebsocket)

if(MSVC)
  target_compile_options(openaxis PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(openaxis PRIVATE -Wall -Wextra -Wpedantic)
endif()
